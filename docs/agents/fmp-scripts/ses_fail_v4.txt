# =====================================================================
# [SES v4] 2026-11-10 ชุด v4: เนื้อหาเหมือน ses_fail v3 ทุกบรรทัด (ถ้าวาง v3 แล้ว ไม่ต้องแก้)
# Script : ses_fail
# Param  : { "code": "...", "message": "...", "snCheck": [ ... ] }
# หน้าที่ : บันทึก ERROR ลง Inbox record ปัจจุบัน แล้วคืน JSON ให้ผู้เรียก
# Version: v2 2026-11-03 (SES) เปลี่ยนบรรทัดเดียว มี "[SES v2]" นำหน้า
# Version: v3 2026-11-10 (SES) คืน "snCheck" ที่สคริปต์ ses ส่งมาด้วย 2 บรรทัด มี "[SES v3]" นำหน้า
# =====================================================================
Set Error Capture [ On ]
Set Variable [ $p ; Value: Get ( ScriptParameter ) ]
Set Variable [ $code ; Value: JSONGetElement ( $p ; "code" ) ]
Set Variable [ $msg ; Value: JSONGetElement ( $p ; "message" ) ]
# [SES v3] ผลตรวจ SN ทีละเครื่องจากสคริปต์ ses (ว่าง = ยังไม่ได้ตรวจ SN)
Set Variable [ $snCheck ; Value: JSONGetElement ( $p ; "snCheck" ) ]
# 
Go to Layout [ “ses” (ses) ; Animation: None ]
# found set เดิมยังอยู่
Set Variable [ $sesID ; Value: ses::ses_id ]
# 
# [SES v2] เดิม = 1 · สคริปต์ ses v2 เหลือ found set 1 record เสมอ แต่กันไว้: ใช้ record ปัจจุบันถ้ามีอย่างน้อย 1
If [ Get ( FoundCount ) ≥ 1 ]
	Set Field [ ses::ses_status ; "ERROR" ]
	Set Field [ ses::ses_error ; $code & ": " & $msg ]
	Set Field [ ses::status_at ; Get ( CurrentHostTimestamp ) ]
	Set Field [ ses::processed_at ; Get ( CurrentHostTimestamp ) ]
	Commit Records/Requests [ Skip data entry validation ; With dialog: Off ]
End If
# 
# [SES v3] เพิ่ม "snCheck" ท้ายผลลัพธ์
Exit Script [ Text Result: JSONSetElement ( "{}" ;     [ "ok" ; False ; JSONBoolean ] ;     [ "code" ; $code ; JSONString ] ;     [ "sesID" ; $sesID ; JSONString ] ;     [ "message" ; $msg ; JSONString ] ;     [ "snCheck" ; If ( IsEmpty ( $snCheck ) ; "[]" ; $snCheck ) ; JSONArray ] ) ]
