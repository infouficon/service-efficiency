# =====================================================================
# Script : ses
# เรียกโดย : FileMaker Data API พร้อม "script":"ses", "script.param":"<SES reference>"
# Param  : SES ID (reference)
# Result : JSON { ok, code, sesID, saleID, lines, message }
# =====================================================================
# 
Set Error Capture [ On ]
Allow User Abort [ Off ]
# 
# ---------- 0. Config ----------
# ตั้งเป็น 0 บน Production เพื่อปฏิเสธข้อมูลทดสอบ (test = true)
Set Variable [ $ALLOW_TEST ; Value: 1 ]
Set Variable [ $SUPPORTED_VERSION ; Value: 1 ]
Set Variable [ $REQUIRED_STATE ; Value: "SENT_TO_CASHIER" ]
# 
Set Variable [ $sesID ; Value: Trim ( Get ( ScriptParameter ) ) ]
If [ IsEmpty ( $sesID ) ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; False ; JSONBoolean ] ;         [ "code" ; "NO_PARAM" ; JSONString ] ;         [ "message" ; "Script parameter (SES ID) is empty" ; JSONString ] ) ]
End If
# 
# ---------- 1. ค้นหา record ของ SES ID นั้นๆ ----------
Go to Layout [ “ses” (ses) ; Animation: None ]
Enter Find Mode [ Pause: Off ]
Set Field [ ses::ses_id ; "==" & Quote ( $sesID ) ]
Perform Find []
# 
If [ Get ( FoundCount ) ≠ 1 ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; False ; JSONBoolean ] ;         [ "code" ; "NOT_FOUND" ; JSONString ] ;         [ "sesID" ; $sesID ; JSONString ] ) ]
End If
# 
# ---------- 2. ตรวจสถานะ (กันรันซ้ำ / รองรับ Reprocess) ----------
If [ ses::ses_status = "DONE" ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; True ; JSONBoolean ] ;         [ "code" ; "ALREADY_DONE" ; JSONString ] ;         [ "sesID" ; $sesID ; JSONString ] ;         [ "saleID" ; ses::sale_id ; JSONString ] ) ]
Else If [ ses::ses_status = "PROCESSING" ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; False ; JSONBoolean ] ;         [ "code" ; "IN_PROGRESS" ; JSONString ] ;         [ "sesID" ; $sesID ; JSONString ] ) ]
End If
# 
# # สถานะว่าง หรือ ERROR → ประมวลผลต่อ
Set Variable [ $json ; Value: ses::ses_body ]
Set Field [ ses::ses_status ; "PROCESSING" ]
Set Field [ ses::ses_error ; "" ]
Set Field [ ses::status_at ; Get ( CurrentHostTimestamp ) ]
Commit Records/Requests [ Skip data entry validation ; With dialog: Off ]
# 
# =====================================================================
# 3. VALIDATION + เตรียมข้อมูล (single-pass loop = try/catch)
#    ผลลัพธ์: ตัวแปร repetition $L_sn[n]
# =====================================================================
# 
Set Variable [ $errCode ; Value: "" ]
Set Variable [ $errMsg ; Value: "" ]
# 
Loop [ Flush: Minimum ]
	# ----- 3.1 โครงสร้าง JSON -----
	If [ Left ( JSONFormatElements ( $json ) ; 1 ) = "?" ]
		Set Variable [ $errCode ; Value: "INVALID_JSON" ]
		Set Variable [ $errMsg ; Value: "ses_body is not valid JSON" ]
		Exit Loop If [ True ]
	End If
	# 
	# ----- 3.2 Header -----
	Set Variable [ $ref ; Value: JSONGetElement ( $json ; "reference" ) ]
	Set Variable [ $version ; Value: JSONGetElement ( $json ; "version" ) ]
	Set Variable [ $state ; Value: JSONGetElement ( $json ; "state" ) ]
	Set Variable [ $isTest ; Value: GetAsBoolean ( JSONGetElement ( $json ; "test" ) ) ]
	Set Variable [ $branch ; Value: Trim ( JSONGetElement ( $json ; "branch.code" ) ) ]
	Set Variable [ $phone ; Value: Trim ( JSONGetElement ( $json ; "customerPhone" ) ) ]
	Set Variable [ $queue ; Value: JSONGetElement ( $json ; "queue" ) ]
	Set Variable [ $staffID ; Value: GetAsNumber ( JSONGetElement ( $json ; "staffId" ) ) ]
	Set Variable [ $source ; Value: JSONGetElement ( $json ; "source" ) ]
	# createdAt "2026-10-29T17:31:31+07:00" → Timestamp (เวลาไทยตามที่ SES ส่งมา)
	Set Variable [ $createdAt ; Value: Let (   [ 	t  = JSONGetElement ( $json ; "createdAt" ) ; 	d  = Date ( Middle ( t ; 6 ; 2 ) ; Middle ( t ; 9 ; 2 ) ; Left ( t ; 4 ) ) ; 	tm = Time ( Middle ( t ; 12 ; 2 ) ; Middle ( t ; 15 ; 2 ) ; Middle ( t ; 18 ; 2 ) )   ] ; 	If ( Length ( t ) ≥ 19 ; T… ]
	# เช็ค SES ID
	If [ $ref ≠ $sesID ]
		Set Variable [ $errCode ; Value: "REFERENCE_MISMATCH" ]
		Set Variable [ $errMsg ; Value: "reference in body ≠ script parameter" ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค version
	If [ GetAsNumber ( $version ) ≠ $SUPPORTED_VERSION ]
		Set Variable [ $errCode ; Value: "UNSUPPORTED_VERSION" ]
		Set Variable [ $errMsg ; Value: "version " & $version & " is not supported" ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค state
	If [ $state ≠ $REQUIRED_STATE ]
		Set Variable [ $errCode ; Value: "INVALID_STATE" ]
		Set Variable [ $errMsg ; Value: "state = " & $state ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค test
	If [ $isTest and not $ALLOW_TEST ]
		Set Variable [ $errCode ; Value: "TEST_NOT_ALLOWED" ]
		Set Variable [ $errMsg ; Value: "Test transaction rejected on this server" ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค เบอร์โทรศัพท์ลูกค้า
	If [ IsEmpty ( $phone ) ]
		Set Variable [ $errCode ; Value: "MISSING_PHONE" ]
		Set Variable [ $errMsg ; Value: "customerPhone is empty" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $phone ; Value: Substitute ( $phone ; [ "-" ; "" ] ; [ " " ; "" ] ) ]
	Set Variable [ $customerID ; Value: Let (   [ 	x = ExecuteSQL ( "select customer_id from customer where cust_tel = ? ORDER BY \"_CreationTimestamp\"" ; "" ; "" ; $phone )   ] ; 	If ( x = "" or x = "?" ; "" ; x )  ) ]
	Set Variable [ $custCnt ; Value: ValueCount ( $customerID ) ]
	If [ $custCnt = 0 ]
		Set Variable [ $errCode ; Value: "WRONG_PHONE" ]
		Set Variable [ $errMsg ; Value: "customerPhone is wrong" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $customer ; Value: GetValue ( $customerID ; $custCnt ) ]
	# 
	# เช็ค สาขา
	If [ IsEmpty ( $branch ) ]
		Set Variable [ $errCode ; Value: "MISSING_BRANCH" ]
		Set Variable [ $errMsg ; Value: "branch.code is empty" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $branchID ; Value: Let (   [ 	x = ExecuteSQL ( "select branch_id from branch where branch_code = ?" ; "" ; "" ; $branch )   ] ; 	If ( x = "" or x = "?" ; "" ; x )  ) ]
	If [ IsEmpty ( $branchID ) ]
		Set Variable [ $errCode ; Value: "WRONG_BRANCH" ]
		Set Variable [ $errMsg ; Value: "branch.code is wrong" ]
		Exit Loop If [ True ]
	End If
	# 
	# ----- 3.3 Payments -----
	If [ JSONGetElementType ( $json ; "payments" ) ≠ JSONArray ]
		Set Variable [ $errCode ; Value: "INVALID_PAYMENTS" ]
		Set Variable [ $errMsg ; Value: "payments must be an array" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $payCnt ; Value: ValueCount ( JSONListKeys ( $json ; "payments" ) ) ]
	Set Variable [ $payby ; Value: JSONListValues ( $json ; "payments" ) ]
	# ตัด ¶ ท้ายสุด (ถ้ามี) และบรรทัดว่าง
	Set Variable [ $payby ; Value: While ( [ r = $payby ] ; Right ( r ; 1 ) = ¶ ; [ r = Left ( r ; Length ( r ) - 1 ) ] ; r ) ]
	Set Variable [ $payby ; Value: Substitute ( $payby ; 		[ "Cash" ; "เงินสด ( Cash )" ] ; 		[ "Transfer" ; "โอนเงิน ( Bank Transfer )" ] ; 		[ "Credit Card" ; "บัตรเครดิต ( Credit Card )" ] ; 		[ "Installment" ; "บัตรเงินผ่อน (Card Installment)" ] ; 		[ "Deposit" ; "ใช้มัดจำ (Deposit)"… ]
	If [ $payCnt = 0 or IsEmpty ( $payby ) ]
		Set Variable [ $errCode ; Value: "NO_PAYMEN" ]
		Set Variable [ $errMsg ; Value: "payments is empty" ]
		Exit Loop If [ True ]
	End If
	# 
	# ----- 3.4 Accessories (ยังไม่มี spec → ไม่ยอมให้ผ่านแบบเงียบ ๆ) -----
	If [ JSONGetElementType ( $json ; "accessories" ) = JSONArray          and ValueCount ( JSONListKeys ( $json ; "accessories" ) ) > 0 ]
		Set Variable [ $errCode ; Value: "ACCESSORIES_NOT_SUPPORTED" ]
		Set Variable [ $errMsg ; Value: "accessories is not empty; handler not implemented yet" ]
		Exit Loop If [ True ]
	End If
	# 
	# ----- 3.5 Items -----
	If [ JSONGetElementType ( $json ; "items" ) ≠ JSONArray ]
		Set Variable [ $errCode ; Value: "INVALID_ITEMS" ]
		Set Variable [ $errMsg ; Value: "items must be an array" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $itemCnt ; Value: ValueCount ( JSONListKeys ( $json ; "items" ) ) ]
	If [ $itemCnt = 0 ]
		Set Variable [ $errCode ; Value: "NO_ITEMS" ]
		Set Variable [ $errMsg ; Value: "items is empty" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $line ; Value: 0 ]
	Set Variable [ $allSN ; Value: "" ]
	Set Variable [ $i ; Value: 0 ]
	# ===== Loop นอก : items (UPC)
	Loop [ Flush: Minimum ]
		Exit Loop If [ $i ≥ $itemCnt ]
		Set Variable [ $item ; Value: JSONGetElement ( $json ; "items[" & $i & "]" ) ]
		Set Variable [ $partNo ; Value: Trim ( JSONGetElement ( $item ; "partNo" ) ) ]
		Set Variable [ $qty ; Value: GetAsNumber ( JSONGetElement ( $item ; "quantity" ) ) ]
		# เช็คจำนวน upc
		If [ IsEmpty ( $partNo ) ]
			Set Variable [ $errCode ; Value: "MISSING_PARTNO" ]
			Set Variable [ $errMsg ; Value: "items[" & $i & "].partNo is empty" ]
		Else If [ JSONGetElementType ( $item ; "sn" ) ≠ JSONArray ]
			Set Variable [ $errCode ; Value: "INVALID_SN" ]
			Set Variable [ $errMsg ; Value: "items[" & $i & "].sn must be an array" ]
		End If
		Exit Loop If [ not IsEmpty ( $errCode ) ]
		# 
		# เช็คจำนวน sn
		Set Variable [ $snArr ; Value: JSONGetElement ( $item ; "sn" ) ]
		Set Variable [ $snCnt ; Value: ValueCount ( JSONListKeys ( $snArr ; "" ) ) ]
		If [ $qty < 1 or $snCnt ≠ $qty ]
			Set Variable [ $errCode ; Value: "SN_COUNT_MISMATCH" ]
			Set Variable [ $errMsg ; Value: $partNo & " quantity=" & $qty & " but sn=" & $snCnt ]
			Exit Loop If [ True ]
		End If
		# 
		Set Variable [ $j ; Value: 0 ]
		# ===== Loop ใน : sn
		Loop [ Flush: Minimum ]
			Exit Loop If [ $j ≥ $snCnt ]
			Set Variable [ $sn ; Value: Trim ( JSONGetElement ( $snArr ; "[" & $j & "]" ) ) ]
			# ดึงข้อมูล SN + UPC + Model + ราคา + ต้นทุน ในครั้งเดียว (ฟิลด์ข้อความที่อาจมี ¶ ปน เช่น product_name ต้องอยู่ท้ายสุด)
			Set Variable [ $snRow ; Value: ExecuteSQL (     "SELECT d.product_upc, d.status, p.product_model, p.Price_Sell,             d.Cost_afterVat, d.Cost_beforeVat, d.vatValue, d.noINV,             p.product_brand, p.product_Category, p.product_SubCategory,             p.product_SubSubCateg… ]
			Set Variable [ $snUPC ; Value: Trim ( GetValue ( $snRow ; 1 ) ) ]
			Set Variable [ $snStatus ; Value: Trim ( GetValue ( $snRow ; 2 ) ) ]
			Set Variable [ $snModel ; Value: Trim ( GetValue ( $snRow ; 3 ) ) ]
			Set Variable [ $snPrice ; Value: GetAsNumber ( GetValue ( $snRow ; 4 ) ) ]
			Set Variable [ $snBranch ; Value: Trim ( GetValue ( $snRow ; 13 ) ) ]
			# 
			# SN ว่าง
			If [ IsEmpty ( $sn ) ]
				Set Variable [ $errCode ; Value: "EMPTY_SN" ]
				Set Variable [ $errMsg ; Value: $partNo & " sn[" & $j & "] is empty" ]
				# SN ซ้ำกันเองใน payload
			Else If [ Position ( ¶ & $allSN & ¶ ; ¶ & $sn & ¶ ; 1 ; 1 ) > 0 ]
				Set Variable [ $errCode ; Value: "DUPLICATE_SN_IN_PAYLOAD" ]
				Set Variable [ $errMsg ; Value: "SN " & $sn & " appears more than once" ]
				# SN error
			Else If [ $snRow = "?" ]
				Set Variable [ $errCode ; Value: "SN_CHECK_ERROR" ]
				Set Variable [ $errMsg ; Value: "Cannot query stock for SN " & $sn ]
				# SN มีในสต๊อกไหม
			Else If [ IsEmpty ( $snRow ) ]
				Set Variable [ $errCode ; Value: "SN_NOT_FOUND" ]
				Set Variable [ $errMsg ; Value: "SN " & $sn & " not found in stock" ]
				# SN อยู่ในสต็อกของสาขาที่เปิดบิลไหม
			Else If [ $snBranch ≠ $branchID ]
				Set Variable [ $errCode ; Value: "SN_WRONG_BRANCH" ]
				Set Variable [ $errMsg ; Value: "SN " & $sn & " is in branch " & $snBranch & ", not " & $branch & " (" & $branchID & ")" ]
				# SN สถานะเป็น ready ไหม
			Else If [ $snStatus ≠ "ready" ]
				Set Variable [ $errCode ; Value: "SN_NOT_READY_FOR_SALE" ]
				Set Variable [ $errMsg ; Value: "SN " & $sn & " status = " & $snStatus ]
				# เช็คว่าใช่ model เดียวกันไหม
			Else If [ $snModel ≠ $partNo ]
				Set Variable [ $errCode ; Value: "SN_MODEL_MISMATCH" ]
				Set Variable [ $errMsg ; Value: "SN " & $sn & " is model " & $snModel & " (UPC " & $snUPC & "), not " & $partNo ]
				# เช็คราคา
			Else If [ IsEmpty ( $snPrice ) or $snPrice ≤ 0 ]
				Set Variable [ $errCode ; Value: "PRICE_NOT_FOUND" ]
				Set Variable [ $errMsg ; Value: "UPC " & $snUPC & " has no selling price" ]
				# เช็คว่าขาย หรือ อยู่ในรายการพักบิลหรือยัง
			Else If [ ExecuteSQL ( "SELECT COUNT(*) FROM POS_front_sale_detail WHERE product_sn = ?               AND ( ( LOWER(invStatus) = 'complete' AND CN_ID IS NULL )                  OR ( invStatus IS NULL AND mark_tmp = 'tmp' ) )" ; "" ; "" ; $sn ) > 0 ]
				Set Variable [ $errCode ; Value: "SN_ALREADY_SOLD_OR_HELD" ]
				Set Variable [ $errMsg ; Value: "SN " & $sn & " is already sold or in a held bill" ]
			End If
			Exit Loop If [ not IsEmpty ( $errCode ) ]
			# 
			# ก็บลงตัวแปร repetition (ไม่ต้อง parse JSON ซ้ำตอนเขียนข้อมูล)
			Set Variable [ $line ; Value: $line + 1 ]
			Set Variable [ $L_sn[$line] ; Value: $sn ]
			Set Variable [ $L_row[$line] ; Value: $snRow ]
			Set Variable [ $allSN ; Value: List ( $allSN ; $sn ) ]
			Set Variable [ $j ; Value: $j + 1 ]
		End Loop
		Exit Loop If [ not IsEmpty ( $errCode ) ]
		# 
		Set Variable [ $i ; Value: $i + 1 ]
	End Loop
	# 
	# จบ single-pass
	Exit Loop If [ True ]
End Loop
# 
If [ not IsEmpty ( $errCode ) ]
	Perform Script [ “ses_fail” ; Specified: From list ; Parameter: JSONSetElement ( "{}" ;             [ "code" ; $errCode ; JSONString ] ;             [ "message" ; $errMsg ; JSONString ] ) ]
	Exit Script [ Text Result: Get ( ScriptResult ) ]
End If
# 
# =====================================================================
# 4. เขียนบิลใน Transaction เดียว (ล้มตรงไหน = Rollback ทั้งบิล)
# =====================================================================
Open Transaction []
	# In Claris FileMaker Pro, custom error codes specified in the Revert Transaction script step must fall within the range of 5000 to 5499.
When you assign a custom code and message within this range, FileMaker stops the transaction, reverts any changes made within the block, and passes your specified code to the Get ( LastError ) function right after the Commit Transaction line.
	# 
	# ----- 4.1 หัวบิล -----
	Go to Layout [ “ขายหน้าร้าน TMP” (pos_front_sale_tmp) ; Animation: None ]
	New Record/Request
	Revert Transaction [ Condition: Get ( LastError ) ≠ 0 ; Error Code: 5001 ; Error Message: "Create sale header failed (" & Get ( LastError ) & ")" ]
	Set Field [ pos_front_sale_tmp::ses_id ; $sesID ]
	Set Field [ pos_front_sale_tmp::ses_queue ; $queue ]
	Set Field [ pos_front_sale_tmp::ses_openTs ; $createdAt ]
	Set Field [ pos_front_sale_tmp::staff_id ; GetAsNumber ( $staffID ) ]
	Set Field [ pos_front_sale_tmp::branch_id ; GetAsNumber ( $branchID ) ]
	Set Field [ pos_front_sale_tmp::branch_idTo ; GetAsNumber ( $branchID ) ]
	Set Field [ pos_front_sale_tmp::customer_id ; $customer ]
	Set Field [ pos_front_sale_tmp::PayBy ; $payby ]
	Set Field [ pos_front_sale_tmp::bill_type ; "S" ]
	Set Field [ pos_front_sale_tmp::status ; "hold" ]
	Revert Transaction [ Condition: Get ( LastError ) ≠ 0 ; Error Code: 5001 ; Error Message: "Create sale header failed (" & Get ( LastError ) & ")" ]
	Set Variable [ $priKey ; Value: pos_front_sale_tmp::_PrimaryKey ]
	Set Variable [ $vatSale ; Value: pos_front_sale_tmp::vat ]
	Set Variable [ $vatType ; Value: pos_front_sale_tmp::vat type ]
	# 
	# ----- 4.2 รายการขาย 1 บรรทัดต่อ 1 SN -----
	Go to Layout [ “POS_front_sale_detail” (POS_front_sale_detail) ; Animation: None ]
	Set Variable [ $n ; Value: 1 ]
	Loop [ Flush: Minimum ]
		Exit Loop If [ $n > $line ]
		New Record/Request
		Revert Transaction [ Condition: Get ( LastError ) ≠ 0 ; Error Code: 5002 ; Error Message: "Create line " & $n & " failed (SN " & $L_sn[$n] & ", err " & Get ( LastError ) & ")" ]
		# 
		Set Variable [ $r ; Value: $L_row[$n] ]
		Set Variable [ $upc ; Value: Trim ( GetValue ( $r ; 1 ) ) ]
		Set Variable [ $old_status ; Value: GetValue ( $r ; 2 ) ]
		Set Variable [ $pdModel ; Value: GetValue ( $r ; 3 ) ]
		Set Variable [ $stPrice ; Value: Let (   [ 	x = GetValue ( $r ; 4 )   ] ; 	If ( x = "" or x = 0 ; 0 ; GetAsNumber ( x ) )  ) ]
		Set Variable [ $cost_a ; Value: Let (   [ 	x = GetValue ( $r ; 5 )   ] ; 	If ( x = "" or x = 0 ; 0 ; GetAsNumber ( x ) )  ) ]
		Set Variable [ $cost_b ; Value: Let (   [ 	x = GetValue ( $r ; 6 )   ] ; 	If ( x = "" or x = 0 ; 0 ; GetAsNumber ( x ) )  ) ]
		Set Variable [ $vat ; Value: Let (   [ 	x = GetValue ( $r ; 7 )   ] ; 	If ( x = "" or x = 0 ; 0 ; GetAsNumber ( x ) )  ) ]
		Set Variable [ $inv ; Value: GetValue ( $r ; 8 ) ]
		Set Variable [ $pdBrand ; Value: GetValue ( $r ; 9 ) ]
		Set Variable [ $pdCat ; Value: GetValue ( $r ; 10 ) ]
		Set Variable [ $pdSub ; Value: GetValue ( $r ; 11 ) ]
		Set Variable [ $pdSubSub ; Value: GetValue ( $r ; 12 ) ]
		Set Variable [ $pdName ; Value: GetValue ( $r ; 14) ]
		# 
		Set Variable [ $title ; Value: TextFormatRemove ( Case ( not IsEmpty ( $pdModel ) ; $pdModel & " " ; "" ) & Case ( not IsEmpty ( $pdName ) ; Substitute ( $pdName ; Char ( 9 ) ; "" ) ; "" ) & ¶ & "SN: " & $L_sn[$n] ) ]
		# 
		Set Variable [ $isLuxury ; Value: "N" ]
		If [ $pdCat = "iPad" ]
			If [ $stPrice ≥ 40000 ]
				Set Variable [ $isLuxury ; Value: "Y" ]
			Else
				Set Variable [ $isLuxury ; Value: "N" ]
			End If
		Else If [ $pdCat = "iPhone" ]
			If [ $stPrice ≥ 40000 ]
				Set Variable [ $isLuxury ; Value: "Y" ]
			Else
				Set Variable [ $isLuxury ; Value: "N" ]
			End If
		Else If [ $pdCat = "Watch" ]
			If [ $stPrice ≥ 40000 ]
				Set Variable [ $isLuxury ; Value: "Y" ]
			Else
				Set Variable [ $isLuxury ; Value: "N" ]
			End If
		Else If [ $pdCat = "Mac" ]
			If [ PatternCount ( $pdSub ; "MACBOOK" ) ]
				If [ $stPrice ≥ 40000 ]
					Set Variable [ $isLuxury ; Value: "Y" ]
				Else
					Set Variable [ $isLuxury ; Value: "N" ]
				End If
			Else
				If [ $stPrice ≥ 100000 ]
					Set Variable [ $isLuxury ; Value: "Y" ]
				Else
					Set Variable [ $isLuxury ; Value: "N" ]
				End If
			End If
		Else If [ $pdCat = "Apple Accy" or $pdCat = "Third Party Accy" ]
			Set Variable [ $isLuxury ; Value: "N" ]
		Else If [ PatternCount ( $pdCat ; "Speaker" ) ]
			Set Variable [ $isLuxury ; Value: "N" ]
		End If
		# 
		Set Field [ POS_front_sale_detail::PrimaryKey ; $priKey ]
		Set Field [ POS_front_sale_detail::sortOrder ; $n ]
		Set Field [ POS_front_sale_detail::mark_tmp ; "tmp" ]
		Set Field [ POS_front_sale_detail::customer_id ; $customer ]
		Set Field [ POS_front_sale_detail::product_sn ; $L_sn[$n] ]
		Set Field [ POS_front_sale_detail::product_upc ; $upc ]
		Set Field [ POS_front_sale_detail::vrt_isLuxury ; $isLuxury ]
		Set Field [ POS_front_sale_detail::store_model ; $pdModel ]
		Set Field [ POS_front_sale_detail::product_name ; Substitute ( $pdName ; Char ( 9 ) ; "" ) ]
		Set Field [ POS_front_sale_detail::store_name ; $pdName ]
		Set Field [ POS_front_sale_detail::pd_Brand ; $pdBrand ]
		Set Field [ POS_front_sale_detail::store_Brand ; $pdBrand ]
		Set Field [ POS_front_sale_detail::pd_category ; $pdCat ]
		Set Field [ POS_front_sale_detail::store_category ; $pdCat ]
		Set Field [ POS_front_sale_detail::pd_SubCat ; If ( not IsEmpty ( $pdSub ) ; $pdSub ; "" ) ]
		Set Field [ POS_front_sale_detail::pd_SubSubCategory ; If ( not IsEmpty ( $pdSubSub ) ; 	$pdSubSub ; "" ) ]
		Set Field [ POS_front_sale_detail::store_SubCat ; If ( not IsEmpty ( $pdSub ) ; $pdSub ; "" ) ]
		Set Field [ POS_front_sale_detail::store_SubSubCat ; If ( not IsEmpty ( $pdSubSub ) ; 	$pdSubSub ; "" ) ]
		Set Field [ POS_front_sale_detail::product_oldStatus ; $old_status ]
		Set Field [ POS_front_sale_detail::vat ; $vatSale ]
		Set Field [ POS_front_sale_detail::vatType ; $vatType ]
		Set Field [ POS_front_sale_detail::product costก่อนvat ; $cost_b ]
		Set Field [ POS_front_sale_detail::cost_beforeVat ; $cost_b ]
		Set Field [ POS_front_sale_detail::cost_afterVat ; $cost_a ]
		Set Field [ POS_front_sale_detail::cost_vatValue ; $vat ]
		Set Field [ POS_front_sale_detail::title ; $title ]
		Set Field [ POS_front_sale_detail::po_inv ; $inv ]
		Set Field [ POS_front_sale_detail::amount ; 1 ]
		Set Field [ POS_front_sale_detail::Type ; "Sale" ]
		Set Field [ POS_front_sale_detail::saleType ; "genSale" ]
		Set Field [ POS_front_sale_detail::SellingPrice ; $stPrice ]
		# 
		Revert Transaction [ Condition: Get ( LastError ) ≠ 0 ; Error Code: 5002 ; Error Message: "Create line " & $n & " failed (SN " & $L_sn[$n] & ", err " & Get ( LastError ) & ")" ]
		Set Variable [ $n ; Value: $n + 1 ]
	End Loop
	# 
	# ----- 4.3 อัปเดต Inbox เป็น DONE (อยู่ใน Transaction เดียวกัน) -----
	Go to Layout [ “ses” (ses) ; Animation: None ]
	Set Field [ ses::ses_status ; "DONE" ]
	Set Field [ ses::sale_id ; $priKey ]
	Set Field [ ses::line_count ; $line ]
	Set Field [ ses::status_at ; Get ( CurrentHostTimestamp ) ]
	Set Field [ ses::processed_at ; Get ( CurrentHostTimestamp ) ]
	Revert Transaction [ Condition: Get ( LastError ) ≠ 0 ; Error Code: 5003 ; Error Message: "Update inbox failed (" & Get ( LastError ) & ")" ]
	# 
Commit Transaction
# 
Set Variable [ $txErr ; Value: Get ( LastError ) ]
Set Variable [ $txDetail ; Value: Get ( LastErrorDetail ) ]
# 
# ---------- 5. ผลลัพธ์ ----------
If [ $txErr ≠ 0 ]
	Perform Script [ “ses_fail” ; Specified: From list ; Parameter: JSONSetElement ( "{}" ;             [ "code" ; "TRANSACTION_FAILED" ; JSONString ] ;             [ "message" ; $txErr & " " & $txDetail ; JSONString ] ) ]
	Exit Script [ Text Result: Get ( ScriptResult ) ]
End If
