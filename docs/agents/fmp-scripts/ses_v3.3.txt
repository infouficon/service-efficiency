# [SES v3] v3.1 (ตามทีม FMP 2026-11-05): เครื่องนั้น "ข้าม" ไม่เขียนบรรทัด (ช่อง SN เป็นช่องหลังบ้าน แคชเชียร์กรอกไม่ได้)

# =====================================================================
# Script : ses
# เรียกโดย : FileMaker Data API พร้อม "script":"ses", "script.param":"<SES reference>"
# Param  : SES ID (reference)
# Result : JSON { ok, code, sesID, saleID, lines, message, warnings, snCheck }
# Version: v2 2026-11-03 แก้โดยทีม SES ให้ตรงกับข้อมูลที่ SES ส่งจริง ทุกจุดที่เปลี่ยนมี "[SES v2]" นำหน้า ที่เหลือคือของเดิม
# Version: v3 2026-11-05 แก้โดยทีม SES เครื่องที่ยังไม่มี SN หรือ FMP ยังไม่มีในสาขานี้ ข้ามไม่เขียน พร้อมจำนวนเครื่องที่ข้ามและเหตุผล
# Version: v3.2 2026-11-06 ตัวที่ทีม FMP ใช้งานจริง + แก้ 3.3 Payments ให้อ่าน "paymentTypes"
# Version: v3.3 2026-11-10 เพิ่ม "snCheck" ในผลลัพธ์ (ผลตรวจ SN ทีละเครื่อง ให้ SES แสดงและส่งกลับให้ Stock สแกนใหม่เฉพาะเครื่องที่ผิด)
# =====================================================================
# 
Set Error Capture [ On ]
Allow User Abort [ Off ]
# 
# ---------- 0. Config ----------
# ตั้งเป็น 0 บน Production เพื่อปฏิเสธข้อมูลทดสอบ (test = true)
Set Variable [ $ALLOW_TEST ; Value: 1 ]
Set Variable [ $SUPPORTED_VERSION ; Value: 1 ]
Set Variable [ $REQUIRED_STATE ; Value: "SENT_TO_CASHIER" ]
# [SES v2] 1 = ลูกค้าใหม่ (ยังไม่มีในตาราง customer) เปิดบิลพักได้โดยยังไม่ผูกลูกค้า แคชเชียร์เลือก/สร้างลูกค้าในบิลเอง · 0 = ปฏิเสธ (WRONG_PHONE) แบบเดิม
Set Variable [ $ALLOW_UNKNOWN_CUSTOMER ; Value: 1 ]
# [SES v2] 1 = ไม่มีวิธีชำระ เปิดบิลพักได้ แคชเชียร์เลือกตอนรับเงิน (SES ไม่บังคับให้เลือกวิธีชำระ) · 0 = ปฏิเสธ (NO_PAYMENT)
Set Variable [ $ALLOW_NO_PAYMENT ; Value: 1 ]
# [SES v2] คำเตือนที่ไม่ทำให้ล้ม จะส่งกลับใน "warnings" และ "message" (SES แสดงในหน้า Admin)
Set Variable [ $warnings ; Value: "" ]
# [SES v3] 1 = เครื่องที่ยังไม่มี SN (SES ส่ง sn น้อยกว่า quantity, "snMissing") หรือ SN ที่ FMP ยังไม่มีในสาขานี้ (SN_NOT_FOUND,
# [SES v3]     SN_WRONG_BRANCH, SN_NOT_READY_FOR_SALE) ข้ามไม่เขียนบรรทัด แคชเชียร์สแกนขายเพิ่มหน้าร้าน · 0 = ปฏิเสธทั้งออเดอร์แบบ v2
# [SES v3] SN รุ่นไม่ตรง / ขายไปแล้ว / อยู่ในบิลพักอื่น / ซ้ำ ยังปฏิเสธเหมือนเดิม (หยิบผิดเครื่อง ต้องให้ Stock สแกนใหม่)
Set Variable [ $ALLOW_SN_LATER ; Value: 1 ]
# [SES v3] จำนวนเครื่องที่ข้าม (แคชเชียร์ต้องสแกนขายเพิ่ม) และเหตุผล (ส่งกลับใน "snPending" และ "message")
Set Variable [ $snPending ; Value: 0 ]
Set Variable [ $snLaterMsg ; Value: "" ]
# [SES v3.3] ผลตรวจ SN ทีละเครื่อง (JSON array) และตัวนับ ส่งกลับใน "snCheck"
Set Variable [ $snCheck ; Value: "[]" ]
Set Variable [ $k ; Value: 0 ]
# 
Set Variable [ $sesID ; Value: Trim ( Get ( ScriptParameter ) ) ]
If [ IsEmpty ( $sesID ) ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; False ; JSONBoolean ] ;         [ "code" ; "NO_PARAM" ; JSONString ] ;         [ "message" ; "Script parameter (SES ID) is empty" ; JSONString ] ) ]
End If
# 
# ---------- 1. ค้นหา record ของ SES ID นั้นๆ ----------
Go to Layout [ “ses” (ses) ; Animation: None ]
Enter Find Mode [ Pause: Off ]
Set Field [ ses::ses_id ; "==" & Quote ( $sesID ) ]
Perform Find []
# 
# [SES v2] เดิม: FoundCount ≠ 1 → NOT_FOUND · ตอนนี้ไม่เจอเลยเท่านั้นที่เป็น NOT_FOUND
If [ Get ( FoundCount ) = 0 ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; False ; JSONBoolean ] ;         [ "code" ; "NOT_FOUND" ; JSONString ] ;         [ "sesID" ; $sesID ; JSONString ] ) ]
End If
# [SES v2] SES เก็บ 1 record ต่อออเดอร์อยู่แล้ว (ส่งซ้ำ = แก้ record เดิม) ถ้ายังเจอมากกว่า 1 ให้ใช้ record ล่าสุด
# (found set ที่ยังไม่ sort เรียงตามลำดับที่สร้าง: เหลือไว้ตัวสุดท้ายตัวเดียว ses_fail จะได้แก้ record ที่ถูกต้อง)
If [ Get ( FoundCount ) > 1 ]
	Go to Record/Request/Page [ First ]
	Omit Multiple Records [ With dialog: Off ; Get ( FoundCount ) - 1 ]
End If
# 
# ---------- 2. ตรวจสถานะ (กันรันซ้ำ / รองรับ Reprocess) ----------
If [ ses::ses_status = "DONE" ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; True ; JSONBoolean ] ;         [ "code" ; "ALREADY_DONE" ; JSONString ] ;         [ "sesID" ; $sesID ; JSONString ] ;         [ "saleID" ; ses::sale_id ; JSONString ] ) ]
Else If [ ses::ses_status = "PROCESSING" ]
	Exit Script [ Text Result: JSONSetElement ( "{}" ;         [ "ok" ; False ; JSONBoolean ] ;         [ "code" ; "IN_PROGRESS" ; JSONString ] ;         [ "sesID" ; $sesID ; JSONString ] ) ]
End If
# 
# # สถานะว่าง หรือ ERROR → ประมวลผลต่อ
Set Variable [ $json ; Value: ses::ses_body ]
Set Field [ ses::ses_status ; "PROCESSING" ]
Set Field [ ses::ses_error ; "" ]
Set Field [ ses::status_at ; Get ( CurrentHostTimestamp ) ]
Commit Records/Requests [ Skip data entry validation ; With dialog: Off ]
# 
# =====================================================================
# 3. VALIDATION + เตรียมข้อมูล (single-pass loop = try/catch)
#    ผลลัพธ์: ตัวแปร repetition $L_sn[n]
# =====================================================================
# 
Set Variable [ $errCode ; Value: "" ]
Set Variable [ $errMsg ; Value: "" ]
# 
Loop [ Flush: Minimum ]
	# ----- 3.1 โครงสร้าง JSON -----
	If [ Left ( JSONFormatElements ( $json ) ; 1 ) = "?" ]
		Set Variable [ $errCode ; Value: "INVALID_JSON" ]
		Set Variable [ $errMsg ; Value: "ses_body is not valid JSON" ]
		Exit Loop If [ True ]
	End If
	# 
	# ----- 3.2 Header -----
	Set Variable [ $ref ; Value: JSONGetElement ( $json ; "reference" ) ]
	Set Variable [ $version ; Value: JSONGetElement ( $json ; "version" ) ]
	Set Variable [ $state ; Value: JSONGetElement ( $json ; "state" ) ]
	Set Variable [ $isTest ; Value: GetAsBoolean ( JSONGetElement ( $json ; "test" ) ) ]
	Set Variable [ $branch ; Value: Trim ( JSONGetElement ( $json ; "branch.code" ) ) ]
	Set Variable [ $phone ; Value: Trim ( JSONGetElement ( $json ; "customerPhone" ) ) ]
	Set Variable [ $queue ; Value: JSONGetElement ( $json ; "queue" ) ]
	Set Variable [ $staffID ; Value: GetAsNumber ( JSONGetElement ( $json ; "staffId" ) ) ]
	Set Variable [ $source ; Value: JSONGetElement ( $json ; "source" ) ]
	# [SES v2] บัญชีที่สร้างเองใน SES (เช่น 0001) ไม่ใช่รหัสพนักงาน ERP: ถ้าไม่พบในตาราง staff ให้เว้นว่าง แคชเชียร์เลือกพนักงานเอง
	# (ถ้า ExecuteSQL error จะได้ "?" ซึ่งไม่ว่าง → ใช้รหัสเดิม ไม่ทำอะไร)
	If [ not IsEmpty ( $staffID ) and IsEmpty ( ExecuteSQL ( "SELECT staff_id FROM staff WHERE staff_id = ?" ; "" ; "" ; $staffID ) ) ]
		Set Variable [ $staffID ; Value: "" ]
		Set Variable [ $warnings ; Value: List ( $warnings ; "UNKNOWN_STAFF" ) ]
	End If
	# [SES v2] บรรทัด $createdAt ด้านล่างถูก FileMaker ตัดตอน export ไม่ต้องแก้ ใช้สูตรเดิมในไฟล์ (SES ส่ง "2026-11-03T14:40:12+07:00")
	# createdAt "2026-10-29T17:31:31+07:00" → Timestamp (เวลาไทยตามที่ SES ส่งมา)
	Set Variable [ $createdAt ; Value: Let (   [ 	t  = JSONGetElement ( $json ; "createdAt" ) ; 	d  = Date ( Middle ( t ; 6 ; 2 ) ; Middle ( t ; 9 ; 2 ) ; Left ( t ; 4 ) ) ; 	tm = Time ( Middle ( t ; 12 ; 2 ) ; Middle ( t ; 15 ; 2 ) ; Middle ( t ; 18 ; 2 ) )   ] ; 	If ( Length ( t ) ≥ 19 ; T… ]
	# เช็ค SES ID
	If [ $ref ≠ $sesID ]
		Set Variable [ $errCode ; Value: "REFERENCE_MISMATCH" ]
		Set Variable [ $errMsg ; Value: "reference in body ≠ script parameter" ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค version
	If [ GetAsNumber ( $version ) ≠ $SUPPORTED_VERSION ]
		Set Variable [ $errCode ; Value: "UNSUPPORTED_VERSION" ]
		Set Variable [ $errMsg ; Value: "version " & $version & " is not supported" ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค state
	If [ $state ≠ $REQUIRED_STATE ]
		Set Variable [ $errCode ; Value: "INVALID_STATE" ]
		Set Variable [ $errMsg ; Value: "state = " & $state ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค test
	If [ $isTest and not $ALLOW_TEST ]
		Set Variable [ $errCode ; Value: "TEST_NOT_ALLOWED" ]
		Set Variable [ $errMsg ; Value: "Test transaction rejected on this server" ]
		Exit Loop If [ True ]
	End If
	# 
	# เช็ค เบอร์โทรศัพท์ลูกค้า
	If [ IsEmpty ( $phone ) ]
		Set Variable [ $errCode ; Value: "MISSING_PHONE" ]
		Set Variable [ $errMsg ; Value: "customerPhone is empty" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $phone ; Value: Substitute ( $phone ; [ "-" ; "" ] ; [ " " ; "" ] ) ]
	Set Variable [ $customerID ; Value: Let (   [ 	x = ExecuteSQL ( "select customer_id from customer where cust_tel = ? ORDER BY \"_CreationTimestamp\"" ; "" ; "" ; $phone )   ] ; 	If ( x = "" or x = "?" ; "" ; x )  ) ]
	Set Variable [ $custCnt ; Value: ValueCount ( $customerID ) ]
	If [ $custCnt = 0 ]
		# [SES v2] ลูกค้าใหม่ของ SES ยังไม่มีใน FMP: เปิดบิลพักได้โดยยังไม่ผูกลูกค้า (ถ้า $ALLOW_UNKNOWN_CUSTOMER = 1)
		If [ $ALLOW_UNKNOWN_CUSTOMER ]
			Set Variable [ $customer ; Value: "" ]
			Set Variable [ $warnings ; Value: List ( $warnings ; "NEW_CUSTOMER" ) ]
		Else
			Set Variable [ $errCode ; Value: "WRONG_PHONE" ]
			Set Variable [ $errMsg ; Value: "customerPhone is wrong" ]
			Exit Loop If [ True ]
		End If
	Else
		Set Variable [ $customer ; Value: GetValue ( $customerID ; $custCnt ) ]
	End If
	# 
	# เช็ค สาขา
	If [ IsEmpty ( $branch ) ]
		Set Variable [ $errCode ; Value: "MISSING_BRANCH" ]
		Set Variable [ $errMsg ; Value: "branch.code is empty" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $branchID ; Value: Let (   [ 	x = ExecuteSQL ( "select branch_id from branch where branch_code = ?" ; "" ; "" ; $branch )   ] ; 	If ( x = "" or x = "?" ; "" ; x )  ) ]
	If [ IsEmpty ( $branchID ) ]
		Set Variable [ $errCode ; Value: "WRONG_BRANCH" ]
		Set Variable [ $errMsg ; Value: "branch.code is wrong" ]
		Exit Loop If [ True ]
	End If
	# 
	# ----- 3.3 Payments -----
	# [SES v2] อ่าน "paymentTypes" (ประเภทตามรายการของ FMP: เงินสด, โอนเงิน, บัตรเครดิต, บัตรเงินผ่อน, ใช้มัดจำ, Trade In, U-Joy, Partner Mall, อื่นๆ)
	# แทน "payments" (ชื่อช่องทางใน SES เช่น 0% Credit Card, PayNext ซึ่ง ADMIN ของ SES เพิ่มได้เรื่อยๆ) ADMIN ของ SES เลือกประเภทให้ทุกช่องทางเอง
	# [SES v3.2] ตัวที่ใช้งานจริง (export 2026-11-06) ยังอ่าน "payments" 3 ที่ในส่วนนี้ (If, $payCnt, $payby): เปลี่ยนเป็น "paymentTypes"
	# [SES v3.2] "payments" คือชื่อช่องทางของ SES (Cash, Transfer, Full Credit Card…) ไม่ตรงกับ Substitute ภาษาไทยด้านล่าง PayBy จึงได้ชื่อดิบ
	If [ JSONGetElementType ( $json ; "paymentTypes" ) ≠ JSONArray ]
		Set Variable [ $errCode ; Value: "INVALID_PAYMENTS" ]
		Set Variable [ $errMsg ; Value: "paymentTypes must be an array" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $payCnt ; Value: ValueCount ( JSONListKeys ( $json ; "paymentTypes" ) ) ]
	Set Variable [ $payby ; Value: JSONListValues ( $json ; "paymentTypes" ) ]
	# ตัด ¶ ท้ายสุด (ถ้ามี) และบรรทัดว่าง
	Set Variable [ $payby ; Value: While ( [ r = $payby ] ; Right ( r ; 1 ) = ¶ ; [ r = Left ( r ; Length ( r ) - 1 ) ] ; r ) ]
	# [SES v2] แปลงประเภทภาษาไทยเป็น PayBy ของ FMP (1 ต่อ 1) แทน Substitute คำอังกฤษเดิมทั้งบรรทัด
	# TODO ทีม FMP: เติม label ของ Trade In / U-Joy / Partner Mall / อื่นๆ ถ้ามีใน FMP (ไม่เติม = ใช้ข้อความเดิมตามที่ SES ส่ง)
	Set Variable [ $payby ; Value: Substitute ( $payby ;
		[ "เงินสด" ; "เงินสด ( Cash )" ] ;
		[ "โอนเงิน" ; "โอนเงิน ( Bank Transfer )" ] ;
		[ "บัตรเครดิต" ; "บัตรเครดิต ( Credit Card )" ] ;
		[ "บัตรเงินผ่อน" ; "บัตรเงินผ่อน (Card Installment)" ] ;
		[ "ใช้มัดจำ" ; "ใช้มัดจำ (Deposit)" ] ) ]
	If [ $payCnt = 0 or IsEmpty ( $payby ) ]
		# [SES v2] SES ไม่บังคับเลือกวิธีชำระ: เปิดบิลพักได้ แคชเชียร์เลือกตอนรับเงิน (ถ้า $ALLOW_NO_PAYMENT = 1) · รหัส error เดิมสะกด NO_PAYMEN
		If [ $ALLOW_NO_PAYMENT ]
			Set Variable [ $payby ; Value: "" ]
			Set Variable [ $warnings ; Value: List ( $warnings ; "NO_PAYMENT" ) ]
		Else
			Set Variable [ $errCode ; Value: "NO_PAYMENT" ]
			Set Variable [ $errMsg ; Value: "paymentTypes is empty" ]
			Exit Loop If [ True ]
		End If
	End If
	# 
	# ----- 3.4 Accessories -----
	# [SES v2] ตกลงกันแล้ว (2026-11-03): หน้าร้านเพิ่มอุปกรณ์เสริมลงบิลเองใน FMP ไม่ต้องตรวจ
	# SES ส่ง "accessories" เป็น [] เสมอ รายการที่ลูกค้าเลือกอยู่ที่ "accessoriesForCashier" [{name, quantity}] ไว้ดูอ้างอิง (ไม่มี SKU ไม่มี SN)
	# 
	# ----- 3.5 Items -----
	If [ JSONGetElementType ( $json ; "items" ) ≠ JSONArray ]
		Set Variable [ $errCode ; Value: "INVALID_ITEMS" ]
		Set Variable [ $errMsg ; Value: "items must be an array" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $itemCnt ; Value: ValueCount ( JSONListKeys ( $json ; "items" ) ) ]
	If [ $itemCnt = 0 ]
		Set Variable [ $errCode ; Value: "NO_ITEMS" ]
		Set Variable [ $errMsg ; Value: "items is empty" ]
		Exit Loop If [ True ]
	End If
	Set Variable [ $line ; Value: 0 ]
	Set Variable [ $allSN ; Value: "" ]
	Set Variable [ $i ; Value: 0 ]
	# ===== Loop นอก : items (UPC)
	Loop [ Flush: Minimum ]
		Exit Loop If [ $i ≥ $itemCnt ]
		Set Variable [ $item ; Value: JSONGetElement ( $json ; "items[" & $i & "]" ) ]
		Set Variable [ $partNo ; Value: Trim ( JSONGetElement ( $item ; "partNo" ) ) ]
		Set Variable [ $qty ; Value: GetAsNumber ( JSONGetElement ( $item ; "quantity" ) ) ]
		# เช็คจำนวน upc
		If [ IsEmpty ( $partNo ) ]
			Set Variable [ $errCode ; Value: "MISSING_PARTNO" ]
			Set Variable [ $errMsg ; Value: "items[" & $i & "].partNo is empty" ]
		Else If [ JSONGetElementType ( $item ; "sn" ) ≠ JSONArray ]
			Set Variable [ $errCode ; Value: "INVALID_SN" ]
			Set Variable [ $errMsg ; Value: "items[" & $i & "].sn must be an array" ]
		End If
		Exit Loop If [ not IsEmpty ( $errCode ) ]
		# 
		# เช็คจำนวน sn
		Set Variable [ $snArr ; Value: JSONGetElement ( $item ; "sn" ) ]
		Set Variable [ $snCnt ; Value: ValueCount ( JSONListKeys ( $snArr ; "" ) ) ]
		# [SES v3] sn น้อยกว่า quantity ได้ถ้า $ALLOW_SN_LATER = 1 (เครื่องที่ขาดข้ามไป แคชเชียร์สแกนขายเพิ่ม) มากกว่า quantity ไม่ได้
		If [ $qty < 1 or $snCnt > $qty or ( $snCnt < $qty and not $ALLOW_SN_LATER ) ]
			Set Variable [ $errCode ; Value: "SN_COUNT_MISMATCH" ]
			Set Variable [ $errMsg ; Value: $partNo & " quantity=" & $qty & " but sn=" & $snCnt ]
			Exit Loop If [ True ]
		End If
		# [SES v3] เครื่องที่ SES ส่งมาโดยยังไม่มี SN
		Set Variable [ $later ; Value: $qty - $snCnt ]
		# 
		Set Variable [ $j ; Value: 0 ]
		# ===== Loop ใน : sn
		Loop [ Flush: Minimum ]
			Exit Loop If [ $j ≥ $snCnt ]
			Set Variable [ $sn ; Value: Trim ( JSONGetElement ( $snArr ; "[" & $j & "]" ) ) ]
			# [SES v2] บรรทัด $snRow ด้านล่างถูก FileMaker ตัดตอน export ไม่ต้องแก้ ใช้ SQL เดิมในไฟล์ (SES ส่ง sn เป็นตัวพิมพ์ใหญ่ ตามที่สแกนจากกล่อง)
